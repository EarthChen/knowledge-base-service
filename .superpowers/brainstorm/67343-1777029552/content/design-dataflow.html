关键接口定义

新增数据类型
class ImportanceTier(Enum):
    CORE = "core"         # Top 20%
    STANDARD = "standard" # 中间 50%
    SKELETON = "skeleton"  # 底部 30%

@dataclass
class CodeSnippet:
    source: str        # 代码原文
    file_path: str
    start_line: int
    end_line: int
    origin: str        # "chunk" | "file" | "signature"

@dataclass
class ChunkSnippet:
    text: str           # 代码片段
    file_path: str
    score: float        # 相似度分数
    parent_name: str    # 所属模块/类名

@dataclass
class EnrichedContext:
    graph_data: PageData
    code_snippets: list[CodeSnippet]
    related_chunks: list[ChunkSnippet]
    related_docs: list[str]
    importance: ImportanceTier
新增/修改接口
class SourceCodeReader:
    async def read(
        node: GraphNode,
        repo_path: str,
        budget_tokens: int = 3000
    ) -> list[CodeSnippet]

class ChunkRetriever:
    async def retrieve(
        entity_name: str,
        entity_fqn: str,
        repository: str,
        k: int = 5,
        exclude_uids: set[str] = None
    ) -> list[ChunkSnippet]

class ImportanceScorer:
    async def score_all(
        repository: str,
        nodes: list[GraphNode]
    ) -> dict[str, ImportanceTier]

class EnrichedContextAssembler:
    async def assemble(
        repository: str,
        node: GraphNode,
        tier: ImportanceTier,
        repo_path: str | None
    ) -> EnrichedContext

class BusinessDomainPlanner:
    async def classify(
        repository: str,
        modules: list[GraphNode]
    ) -> dict[str, list[str]]
配置新增 
# .env 新增
WIKI__REPO_BASE_PATH=/path/to/repos        # 源码仓库根路径
WIKI__CODE_TOKEN_BUDGET=3000                # 默认代码 token 预算
WIKI__IMPORTANCE_CORE_PERCENTILE=80         # Top 20% 为 core
WIKI__IMPORTANCE_STANDARD_PERCENTILE=30     # 底部 30% 为 skeleton
WIKI__ENRICHMENT_ENABLED=true               # 异步丰富开关
WIKI__BUSINESS_DOMAIN_ENABLED=false         # 业务领域模式开关（Phase3）
